# Redis Add-on AppArmor Profile - 简化版本
profile redis {
  # 基础权限
  #include <abstractions/base>
  #include <abstractions/nameservice>

  # 关键能力
  capability dac_override,
  capability net_bind_service,

  # 宽松的文件访问权限
  /data/** rwk,
  /opt/** rwk,
  /tmp/** rwk,
  /etc/** r,
  /lib/** mr,
  /usr/** mr,
  /proc/** r,
  /dev/** r,

  # 执行权限
  /run.sh ix,
  /usr/bin/redis-server ixr,
  /usr/bin/redis-cli ixr,
  /bin/bash ix,
  /usr/bin/jq ix,
  /usr/bin/gosu ix,

  # 网络访问
  network tcp,
  network udp,

  # 允许所有子进程
  /usr/bin/* ixr,
  /bin/* ixr,
}
